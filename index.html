function getSearchVolumeData(city, suburb, trade) { const tradeKey = trade.toLowerCase(); const base = TRADE_SEARCH_BASE[tradeKey] || TRADE_SEARCH_BASE["plumbing"]; const pop = SUBURB_POPULATION[suburb] || 4000; // Formula: city base * suburb factor * (pop/5000) - gives realistic suburb searches // Winter Valley plumber: 880 * 0.12 * (4200/5000) = ~89 searches/mo let monthlySearches = Math.round(base.cityBase * base.suburbFactor * (pop / 5000)); // Keep it realistic - min 30, max 350 for a suburb monthlySearches = Math.max(30, Math.min(350, monthlySearches + Math.floor(Math.random()*20))); const potentialCalls = Math.round(monthlySearches * base.callRate); // 22% call const potentialJobs = Math.round(potentialCalls * 0.65); // 65% of calls become jobs const revenueMin = potentialJobs * Math.round(base.avgJobValue * 0.7); const revenueMax = potentialJobs * base.avgJobValue; const missedCallsIfNotTop3 = Math.round(potentialCalls * 0.75); // 75% go to Top 3 return { monthlySearches, potentialCalls, potentialJobs, revenueMin, revenueMax, missedCallsIfNotTop3, avgJobValue: base.avgJobValue };
